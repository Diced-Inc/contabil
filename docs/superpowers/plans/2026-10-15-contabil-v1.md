# Contabil v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Sistema contábil web para um contador com várias empresas: importa OFX, sugere classificação (regras + IA), contador revisa, gera lançamentos, fecha período e emite Balanço, DRE, Balancete, Diário e Razão.

**Architecture:** Next.js 16 App Router com Server Actions; Postgres (Supabase) via Drizzle, com triggers garantindo partidas balanceadas e período fechado; toda escrita contábil passa por `src/lib/lancamentos.ts`; relatórios montados em TS a partir de uma única query de saldos. Spec completo em `docs/superpowers/specs/2026-10-15-contabil-design.md` — leia antes de qualquer task.

**Tech Stack:** Next.js 16, TypeScript, Tailwind v4, shadcn/ui, lucide-react, Drizzle ORM + postgres-js, drizzle-kit, Supabase Auth (`@supabase/ssr`), `@anthropic-ai/sdk`, zod, Vitest, pnpm, Node 24.

**Regras para quem executa:**
- Idioma da UI e das mensagens: português do Brasil. Sem emoji; ícones são `lucide-react`.
- Sem `DATABASE_URL` no `.env`? Gere a migration (`pnpm db:generate`) e siga; não bloqueie. Marque no relatório que `db:migrate` ficou pendente.
- Commit ao fim de cada task com a mensagem indicada. Rodar `pnpm test` e `pnpm tsc --noEmit` antes de cada commit.
- Nada de lógica contábil em componente React. Server action valida com zod → chama `src/lib/*` → `revalidatePath`.
- Dinheiro sempre `numeric(15,2)` no banco e `string` no Drizzle; converter com `Number()` só em cálculo/exibição. Nos cálculos em TS use centavos inteiros (`Math.round(x*100)`) para somar.

```sql
-- partidas: conta analítica e da mesma empresa do lançamento
CREATE OR REPLACE FUNCTION check_partida_conta() RETURNS trigger AS $$
DECLARE t tipo_conta; e_conta uuid; e_lanc uuid;
BEGIN
  SELECT tipo, empresa_id INTO t, e_conta FROM plano_contas WHERE id = NEW.conta_id;
  SELECT empresa_id INTO e_lanc FROM lancamentos WHERE id = NEW.lancamento_id;
  IF t IS DISTINCT FROM 'analitica' THEN RAISE EXCEPTION 'Conta % não é analítica', NEW.conta_id; END IF;
  IF e_conta <> e_lanc THEN RAISE EXCEPTION 'Conta pertence a outra empresa'; END IF;
  IF NEW.debito < 0 OR NEW.credito < 0 OR (NEW.debito > 0) = (NEW.credito > 0) THEN
    RAISE EXCEPTION 'Partida precisa ter débito OU crédito positivo';
  END IF;
  RETURN NEW;
END $$ LANGUAGE plpgsql;
CREATE TRIGGER partida_conta BEFORE INSERT OR UPDATE ON partidas
  FOR EACH ROW EXECUTE FUNCTION check_partida_conta();

-- partidas: lançamento balanceado (deferred, avaliado no commit)
CREATE OR REPLACE FUNCTION check_lancamento_balanceado() RETURNS trigger AS $$
DECLARE lid uuid; d numeric; c numeric; n int;
BEGIN
  lid := COALESCE(NEW.lancamento_id, OLD.lancamento_id);
  SELECT coalesce(sum(debito),0), coalesce(sum(credito),0), count(*) INTO d, c, n FROM partidas WHERE lancamento_id = lid;
  IF n = 0 THEN RETURN NULL; END IF; -- lançamento apagado em cascade
  IF n < 2 OR d <> c THEN
    RAISE EXCEPTION 'Lançamento % desbalanceado: D=% C=% (% partidas)', lid, d, c, n;
  END IF;
  RETURN NULL;
END $$ LANGUAGE plpgsql;
CREATE CONSTRAINT TRIGGER lancamento_balanceado AFTER INSERT OR UPDATE OR DELETE ON partidas
  DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION check_lancamento_balanceado();

-- período fechado
CREATE OR REPLACE FUNCTION periodo_fechado(e uuid, d date) RETURNS boolean AS $$
  SELECT EXISTS (SELECT 1 FROM fechamentos WHERE empresa_id = e AND competencia = date_trunc('month', d)::date);
$$ LANGUAGE sql STABLE;

CREATE OR REPLACE FUNCTION check_lancamento_periodo() RETURNS trigger AS $$
BEGIN
  IF TG_OP = 'DELETE' THEN
    IF periodo_fechado(OLD.empresa_id, OLD.data) THEN RAISE EXCEPTION 'Período %/% está fechado', to_char(OLD.data,'MM'), to_char(OLD.data,'YYYY'); END IF;
    RETURN OLD;
  END IF;
  IF TG_OP = 'INSERT' THEN
    IF periodo_fechado(NEW.empresa_id, NEW.data) THEN RAISE EXCEPTION 'Período %/% está fechado', to_char(NEW.data,'MM'), to_char(NEW.data,'YYYY'); END IF;
    RETURN NEW;
  END IF;
  -- UPDATE: só bloqueia se mudar conteúdo (status/estorno_de podem mudar)
  IF (NEW.data, NEW.competencia, NEW.historico, NEW.documento, NEW.data_pagamento)
     IS DISTINCT FROM (OLD.data, OLD.competencia, OLD.historico, OLD.documento, OLD.data_pagamento) THEN
    IF periodo_fechado(OLD.empresa_id, OLD.data) OR periodo_fechado(NEW.empresa_id, NEW.data) THEN
      RAISE EXCEPTION 'Período fechado';
    END IF;
  END IF;
  RETURN NEW;
END $$ LANGUAGE plpgsql;
CREATE TRIGGER lancamento_periodo BEFORE INSERT OR UPDATE OR DELETE ON lancamentos
  FOR EACH ROW EXECUTE FUNCTION check_lancamento_periodo();

CREATE OR REPLACE FUNCTION check_partida_periodo() RETURNS trigger AS $$
DECLARE l lancamentos%ROWTYPE;
BEGIN
  SELECT * INTO l FROM lancamentos WHERE id = COALESCE(NEW.lancamento_id, OLD.lancamento_id);
  IF l.id IS NOT NULL AND periodo_fechado(l.empresa_id, l.data) THEN RAISE EXCEPTION 'Período fechado'; END IF;
  RETURN COALESCE(NEW, OLD);
END $$ LANGUAGE plpgsql;
CREATE TRIGGER partida_periodo BEFORE INSERT OR UPDATE OR DELETE ON partidas
  FOR EACH ROW EXECUTE FUNCTION check_partida_periodo();

CREATE INDEX partidas_conta_idx ON partidas (conta_id);
CREATE INDEX partidas_lanc_idx ON partidas (lancamento_id);
CREATE INDEX lancamentos_empresa_data_idx ON lancamentos (empresa_id, data);
CREATE INDEX transacoes_status_idx ON transacoes_bancarias (conta_bancaria_id, status);
```

Nota: `ON DELETE CASCADE` de `lancamentos` → `partidas` dispara `partida_periodo` com `l.id` nulo (linha já apagada) e passa; o bloqueio nesse caso é do trigger em `lancamentos`.

```ts
export type OfxTransacao = { fitid: string; data: string; valor: string; memo: string; tipo: string | null };
export type OfxExtrato = {
  bankId: string; acctId: string; dtStart: string | null; dtEnd: string | null;
  saldoFinal: string | null; dataSaldo: string | null; transacoes: OfxTransacao[];
};

function decode(buf: Buffer): string {
  const head = buf.subarray(0, 600).toString("latin1");
  const utf8 = /ENCODING:UTF-8|encoding="UTF-8"|CHARSET:UTF-8/i.test(head);
  return buf.toString(utf8 ? "utf8" : "latin1");
}
function tag(src: string, name: string): string | null {
  const m = src.match(new RegExp(`<${name}>([^<\\r\\n]*)`, "i"));
  return m ? m[1].trim() : null;
}
function parseDate(s: string | null): string | null {
  if (!s) return null;
  const m = s.match(/^(\d{4})(\d{2})(\d{2})/);
  return m ? `${m[1]}-${m[2]}-${m[3]}` : null;
}
function parseAmount(s: string | null): string {
  if (!s) throw new Error("TRNAMT ausente");
  const n = Number(s.replace(/\s/g, "").replace(/\.(?=\d{3},)/g, "").replace(",", "."));
  if (Number.isNaN(n)) throw new Error(`Valor inválido: ${s}`);
  return n.toFixed(2);
}

export function parseOfx(buf: Buffer): OfxExtrato {
  const src = decode(buf);
  if (/<CCSTMTRS>/i.test(src)) throw new Error("Extrato de cartão de crédito não é suportado nesta versão");
  const body = src.match(/<STMTRS>([\s\S]*?)(<\/STMTRS>|$)/i)?.[1];
  if (!body) throw new Error("Arquivo não contém extrato bancário (STMTRS)");
  const bankId = tag(body, "BANKID"), acctId = tag(body, "ACCTID");
  if (!bankId || !acctId) throw new Error("BANKID/ACCTID ausentes");
  const blocks = body.match(/<STMTTRN>[\s\S]*?(?=<STMTTRN>|<\/BANKTRANLIST>|<LEDGERBAL>|$)/gi) ?? [];
  if (blocks.length === 0) throw new Error("Nenhuma transação encontrada");
```

```ts
export type PartidaInput = { contaId: string; debito?: string; credito?: string };
export type LancamentoInput = {
  empresaId: string; data: string; competencia?: string; dataPagamento?: string | null;

export function validarPartidas(ps: PartidaInput[]) {
  if (ps.length < 2) throw new Error("Lançamento precisa de pelo menos duas partidas");
  let d = 0, c = 0;
  for (const p of ps) {
    const dd = cents(p.debito ?? 0), cc = cents(p.credito ?? 0);
    if (dd < 0 || cc < 0 || (dd > 0) === (cc > 0)) throw new Error("Cada partida precisa de débito OU crédito positivo");
    d += dd; c += cc;
  }
  if (d !== c) throw new Error(`Lançamento desbalanceado: débitos ${fromCents(d)} ≠ créditos ${fromCents(c)}`);
  return { total: d };
}
```

```ts
import { somenteDigitos, fmtValor, fmtData } from "@/lib/format";
export type RegraLike = { id: string; tipoMatch: "memo_contem" | "documento" | "valor_exato"; padrao: string; sinal: "entrada" | "saida" | "qualquer"; contaId: string; historicoTemplate: string; prioridade: number; ativa: boolean; createdAt: Date };
export type TransacaoLike = { memo: string; valor: string; data: string };

export function casaRegra(r: RegraLike, t: TransacaoLike): boolean {
  if (!r.ativa) return false;
  const v = Number(t.valor);
  if (r.sinal === "entrada" && v <= 0) return false;
  if (r.sinal === "saida" && v >= 0) return false;
  switch (r.tipoMatch) {
    case "memo_contem": return t.memo.toLowerCase().includes(r.padrao.toLowerCase());
    case "documento": return somenteDigitos(t.memo).includes(somenteDigitos(r.padrao));
    case "valor_exato": return Math.abs(v).toFixed(2) === Number(r.padrao.replace(",", ".")).toFixed(2);
  }
}
export function aplicarTemplate(tpl: string, t: TransacaoLike) {
  return tpl.replace("{memo}", t.memo).replace("{valor}", fmtValor(Math.abs(Number(t.valor)))).replace("{data}", fmtData(t.data));
}
export function primeiraRegra(regras: RegraLike[], t: TransacaoLike) {
  const ordenadas = [...regras].sort((a, b) => a.prioridade - b.prioridade || a.createdAt.getTime() - b.createdAt.getTime());
  return ordenadas.find((r) => casaRegra(r, t)) ?? null;
}
```

```ts
export type TxCand = { id: string; contaBancariaId: string; data: string; valor: string };
const dias = (a: string, b: string) => Math.abs((Date.parse(a) - Date.parse(b)) / 86400000);
/** Pares [saída, entrada] entre contas diferentes, mesmo valor invertido, até 2 dias. Cada transação entra em no máximo um par. */
export function parearTransferencias(txs: TxCand[]): Array<[string, string]> {
  const usados = new Set<string>(); const pares: Array<[string, string]> = [];
  const saidas = txs.filter((t) => Number(t.valor) < 0), entradas = txs.filter((t) => Number(t.valor) > 0);
  for (const s of saidas) {
    const e = entradas.find((x) => !usados.has(x.id) && x.contaBancariaId !== s.contaBancariaId
      && Math.abs(Number(x.valor)) === Math.abs(Number(s.valor)) && dias(x.data, s.data) <= 2);
    if (e) { usados.add(s.id); usados.add(e.id); pares.push([s.id, e.id]); }
  }
  return pares;
}
```

```ts
import { cents, fromCents } from "./format";
export type SaldoConta = { id: string; codigo: string; saldo: string }; // saldo = D − C
export type PartidaApuracao = { contaId: string; debito?: string; credito?: string };
/** Zera contas de resultado (classe 3). Lucro (Σ saldo < 0) → crédito em lucros; prejuízo → débito em prejuízos. */
export function partidasApuracao(resultado: SaldoConta[], lucrosId: string, prejuizosId: string): { partidas: PartidaApuracao[]; resultado: string } {
  const partidas: PartidaApuracao[] = []; let total = 0;
  for (const c of resultado.filter((c) => c.codigo.startsWith("3"))) {
    const s = cents(c.saldo); if (s === 0) continue; total += s;
    partidas.push(s > 0 ? { contaId: c.id, credito: fromCents(s) } : { contaId: c.id, debito: fromCents(-s) });
  }
  if (partidas.length === 0) return { partidas: [], resultado: "0.00" };
  const lucro = -total; // saldo credor líquido = lucro
  partidas.push(lucro > 0 ? { contaId: lucrosId, credito: fromCents(lucro) } : { contaId: prejuizosId, debito: fromCents(-lucro) });
  return { partidas, resultado: fromCents(lucro) };
}
```

```ts
import { sql } from "drizzle-orm";
import { db } from "@/db";
export type LinhaSaldo = { id: string; codigo: string; nome: string; natureza: "D" | "C"; tipo: "sintetica" | "analitica"; parentId: string | null; grupoDre: string | null; saldoAnterior: string; debitos: string; creditos: string; saldoAtual: string };
/** Saldo = Σdébito − Σcrédito. anterior: data < de; movimento: de ≤ data ≤ ate. Inclui todas as contas (sintéticas com zeros; hierarquia.ts agrega). */
export async function saldos(empresaId: string, de: string, ate: string): Promise<LinhaSaldo[]> {
  const rows = await db.execute(sql`
    SELECT pc.id, pc.codigo, pc.nome, pc.natureza, pc.tipo, pc.parent_id AS "parentId", pc.grupo_dre AS "grupoDre",
      coalesce(sum(CASE WHEN l.data < ${de} THEN p.debito - p.credito END), 0)::text AS "saldoAnterior",
      coalesce(sum(CASE WHEN l.data >= ${de} AND l.data <= ${ate} THEN p.debito END), 0)::text AS debitos,
      coalesce(sum(CASE WHEN l.data >= ${de} AND l.data <= ${ate} THEN p.credito END), 0)::text AS creditos
    FROM plano_contas pc
    LEFT JOIN partidas p ON p.conta_id = pc.id
    LEFT JOIN lancamentos l ON l.id = p.lancamento_id AND l.data <= ${ate}
    WHERE pc.empresa_id = ${empresaId}
    GROUP BY pc.id ORDER BY pc.codigo`);
  return (rows as unknown as Omit<LinhaSaldo, "saldoAtual">[]).map((r) => ({
    ...r, saldoAtual: (Number(r.saldoAnterior) + Number(r.debitos) - Number(r.creditos)).toFixed(2),
  }));
}
```
